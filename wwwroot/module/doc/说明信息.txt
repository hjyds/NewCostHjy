  Cursor c_Mrpcfa(v_频次 诊疗频率时间.执行频率%Type) Is --默认频次方案
    Select * From 诊疗频率时间 Where 执行频率 = v_频次 Order By 方案序号;

Begin
  ----读取入参
  Select Trim(Extractvalue(Value(A), 'IN/PATIID')) Into v_Patiid From Table(Xmlsequence(Extract(Strxml, 'IN'))) A;
  Select Trim(Extractvalue(Value(A), 'IN/PAGEID')) Into v_Pageid From Table(Xmlsequence(Extract(Strxml, 'IN'))) A;
  Select Trim(Extractvalue(Value(A), 'IN/BABY')) Into v_Baby From Table(Xmlsequence(Extract(Strxml, 'IN'))) A;
  Select Trim(Extractvalue(Value(A), 'IN/ID')) Into v_医嘱id From Table(Xmlsequence(Extract(Strxml, 'IN'))) A;
  Select Trim(Extractvalue(Value(A), 'IN/XX')) Into v_Xx From Table(Xmlsequence(Extract(Strxml, 'IN'))) A;
  Select Trim(Extractvalue(Value(A), 'IN/QX')) Into v_Qx From Table(Xmlsequence(Extract(Strxml, 'IN'))) A;
  Select Trim(Extractvalue(Value(A), 'IN/ZD')) Into v_站点 From Table(Xmlsequence(Extract(Strxml, 'IN'))) A;
  Select To_Number(Extractvalue(Value(A), 'IN/KSID')) Into n_科室id From Table(Xmlsequence(Extract(Strxml, 'IN'))) A;
  Select Trim(Extractvalue(Value(A), 'IN/STARTPAGE'))
  Into v_Startpage
  From Table(Xmlsequence(Extract(Strxml, 'IN'))) A;
  Select Trim(Extractvalue(Value(A), 'IN/PAGECOUNT'))
  Into v_Pagecount
  From Table(Xmlsequence(Extract(Strxml, 'IN'))) A;
  Select Trim(Extractvalue(Value(A), 'IN/ITEMCOUNT'))
  Into v_Itemcount
  From Table(Xmlsequence(Extract(Strxml, 'IN'))) A;

  --定义XML
  x_Templet := Xmltype('<OUTPUT><YZLIST></YZLIST><ZXPC></ZXPC><ZXKS></ZXKS><GYTJ></GYTJ><CJFF></CJFF></OUTPUT>');

  --1、期效=0，v_Xx=在用，走现有长嘱在用分支
  --2、期效=1，v_Xx=今日，走现有临嘱今日分支
  --3、期效=0，v_Xx<>在用,根据v_Xx的值设定时间条件，修改游标查询时间范围，走长嘱非在用分支
  --4、期效=1，v_Xx<>今日,根据v_Xx的值设定时间条件，修改游标查询时间范围，走临嘱非今日分支
  --5、期效=1，v_Xx=在用，返回数据为空
  --6、除新增条件外，保持时间参数范围不变，以兼容之前版本

  Case
    When v_Xx = '今日' Then
      v_Begindate := Trunc(Sysdate);
      v_Enddate   := Sysdate;
    When v_Xx = '昨日' Then
      v_Begindate := Trunc(Sysdate) - 1;
      v_Enddate   := Trunc(Sysdate) - 1 / 24 / 60 / 60;
    When v_Xx = '一周内' Then
      v_Begindate := Trunc(Sysdate) - 7;
      v_Enddate   := Sysdate;
    When v_Xx = '在用' And v_Qx = '临嘱' Then
      v_Begindate := Sysdate + 1;
      v_Enddate   := Sysdate + 1 + 1 / 24 / 60 / 60;
    Else
      --全部
      Select 入院日期 Into v_Begindate From 病案主页 Where 病人id = v_Patiid And 主页id = v_Pageid;
      v_Enddate := Sysdate;
  End Case;

  /*  v_Begindate := Trunc(Sysdate);
  v_Enddate   := Sysdate;*/

  If v_医嘱id Is Not Null Then
  
    Select 参数值 Into v_按规格下医嘱 From zlParameters Where 参数名 = '药品按规格下医嘱';
    If v_Qx = '长嘱' Then
      v_医嘱期效 := 0;
    Else
      v_医嘱期效 := 1;
    End If;
  
    v_执行频次 := '';
    For r_Zxpc In c_Zxpc Loop
      v_执行频次 := '<ITEM jsonArray=''true''><BM>' || r_Zxpc.编码 || '</BM><MC>' || r_Zxpc.名称 || '</MC><JM>' || r_Zxpc.英文名称 ||
                '</JM><PLCS>' || r_Zxpc.频率次数 || '</PLCS><PLJG>' || r_Zxpc.频率间隔 || '</PLJG><JGDW>' || r_Zxpc.间隔单位 ||
                '</JGDW><SYFW>' || r_Zxpc.适用范围 || '</SYFW></ITEM>';
      Select Appendchildxml(x_Templet, '/OUTPUT/ZXPC', Xmltype(v_执行频次)) Into x_Templet From Dual;
    End Loop;
  
    --提取药品默认方案
    v_默认方案 := '';
    For r_Zxpc In c_Ypzxpc Loop
      For r_Mrpcfa In c_Mrpcfa(r_Zxpc.编码) Loop
        v_默认方案 := v_默认方案 || '<ITEM jsonArray=''true''><BM>' || r_Zxpc.编码 || '</BM><SJFA>' || r_Mrpcfa.时间方案 ||
                  '</SJFA><GYTJID>' || r_Mrpcfa.给药途径id || '</GYTJID></ITEM>';
      End Loop;
    End Loop;
    v_默认方案 := '<MRPCFA>' || v_默认方案 || '</MRPCFA>';
    Select Appendchildxml(x_Templet, '/OUTPUT/ZXPC', Xmltype(v_默认方案)) Into x_Templet From Dual;
  
    v_给药途径 := '';
    For r_Gytj In c_Gytj Loop
      v_给药途径 := '<ITEM jsonArray=''true''><ID>' || r_Gytj.Id || '</ID><MC>' || r_Gytj.名称 || '</MC><ZXFL>' ||
                r_Gytj.执行分类 || '</ZXFL><PYJM>' || r_Gytj.简码 || '</PYJM></ITEM>';
      v_Temp     := v_给药途径;
      Select Appendchildxml(x_Templet, '/OUTPUT/GYTJ', Xmltype(v_Temp)) Into x_Templet From Dual;
      v_给药执行 := '';
      Select 执行科室 Into v_执行性质 From 诊疗项目目录 Where ID = r_Gytj.Id;
      If v_执行性质 = 0 Then
        v_给药执行 := '<ITEM jsonArray=''true''><ID>0</ID><MC>' || '无执行叮嘱' || '</MC></ITEM>';
      Elsif v_执行性质 = 5 Then
        v_给药执行 := '<ITEM jsonArray=''true''><ID>0</ID><MC>' || '院外执行' || '</MC></ITEM>';
      Elsif v_执行性质 = 2 Then
        Select b.Id, b.名称
